## Instalação

```bash
npm install @blips/ui
```

## Uso

```tsx
import {
  Popover,
  PopoverContent,
  PopoverDescription,
  PopoverHeader,
  PopoverTitle,
  PopoverTrigger,
} from "@blips/ui/components/popover"
```

```tsx
<Popover>
  <PopoverTrigger asChild>
    <Button variant="outline">Abrir Popover</Button>
  </PopoverTrigger>
  <PopoverContent>
    <PopoverHeader>
      <PopoverTitle>Título</PopoverTitle>
      <PopoverDescription>Texto de descrição aqui.</PopoverDescription>
    </PopoverHeader>
  </PopoverContent>
</Popover>
```
