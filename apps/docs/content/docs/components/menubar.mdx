---
title: Menubar
description: Um menu visualmente persistente, comum em aplicações desktop, que fornece acesso rápido a um conjunto consistente de comandos.
---

<ComponentPreview name="menubar-demo" />

## Instalação

```bash
npm install @blips/ui
```

## Uso

```tsx
import {
  Menubar,
  MenubarContent,
  MenubarGroup,
  MenubarItem,
  MenubarMenu,
  MenubarSeparator,
  MenubarShortcut,
  MenubarTrigger,
} from "@blips/ui/components/menubar"
```

```tsx
<Menubar>
  <MenubarMenu>
    <MenubarTrigger>Arquivo</MenubarTrigger>
    <MenubarContent>
      <MenubarGroup>
        <MenubarItem>
          Nova aba <MenubarShortcut>⌘T</MenubarShortcut>
        </MenubarItem>
        <MenubarItem>Nova janela</MenubarItem>
      </MenubarGroup>
      <MenubarSeparator />
      <MenubarGroup>
        <MenubarItem>Compartilhar</MenubarItem>
        <MenubarItem>Imprimir</MenubarItem>
      </MenubarGroup>
    </MenubarContent>
  </MenubarMenu>
</Menubar>
```
